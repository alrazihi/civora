### 3.2 Metric Structs

All metric structs implement the following interface contract:

- **OrganizationID** — tenant scope (UUID)
- **Period** — time aggregation period
- **Bucket** — time bucket start (UTC)
- **Explicit definition** — each metric has a clear business meaning
- **Authoritative source** — documented source tables
- **Time semantics** — bucket-based time bucketing via `bucketStart()`
- **Deterministic calculation** — pure SQL aggregation, no randomness

### 4.2 Time Semantics

The `bucketStart()` function normalizes time bucketing:

```go
func bucketStart(t time.Time, period MetricPeriod) time.Time {
    t = t.UTC().Truncate(time.Hour)
    switch period {
    case MetricPeriodDaily:
        return time.Date(t.Year(), t.Month(), t.Day(), 0, 0, 0, 0, time.UTC)
    case MetricPeriodWeekly:
        // ISO week start (Monday)
        weekday := int(t.Weekday())
        if weekday == 0 {
            weekday = 7
        }
        return t.AddDate(0, 0, -(weekday - 1)).Truncate(time.Hour).Add(-time.Duration(weekday-1) * 24 * time.Hour)
    case MetricPeriodMonthly:
        return time.Date(t.Year(), t.Month(), 1, 0, 0, 0, 0, time.UTC)
    default:
        return t
    }
}
```

All queries use `bucketStart()` to ensure consistent time bucketing across daily, weekly, and monthly periods.

### 4.3 PostgreSQL Queries

All queries are parameterized and tenant-scoped:

**GetCaseVolume:**
```sql
SELECT
    COUNT(*) AS total_cases,
    COUNT(*) FILTER (WHERE status = 'NEW') AS new_cases,
    COUNT(*) FILTER (WHERE status = 'CLOSED') AS closed_cases,
    COUNT(*) FILTER (WHERE status = 'REJECTED') AS rejected_cases
FROM cases
WHERE organization_id = $1
  AND created_at >= bucket_start
  AND created_at < bucket_start + interval
```

**GetWorkflowThroughput:**
```sql
SELECT
    ws.name AS from_state,
    ws2.name AS to_state,
    COUNT(*) AS transition_count
FROM workflow_transition_history wth
JOIN workflow_instances wi ON wth.workflow_instance_id = wi.id
JOIN workflow_states ws ON wth.from_state_id = ws.id
JOIN workflow_states ws2 ON wth.to_state_id = ws2.id
WHERE wi.organization_id = $1
  AND wth.created_at >= bucket_start
  AND wth.created_at < bucket_start + interval
GROUP BY ws.name, ws2.name
```

**GetStateDuration:**
```sql
SELECT
    ws.name AS state_name,
    AVG(EXTRACT(EPOCH FROM (wth2.created_at - wth.created_at)) / 60) AS avg_duration_minutes,
    COUNT(*) AS case_count
FROM workflow_transition_history wth
JOIN workflow_states ws ON wth.to_state_id = ws.id
JOIN workflow_transition_history wth2 ON wth2.workflow_instance_id = wth.workflow_instance_id
    AND wth2.id > wth.id
WHERE wth.workflow_instance_id IN (
    SELECT id FROM workflow_instances WHERE organization_id = $1
)
AND wth.created_at >= bucket_start
AND wth.created_at < bucket_start + interval
GROUP BY ws.name
```

**GetCaseCycleTime:**
```sql
SELECT
    AVG(EXTRACT(EPOCH FROM (closed_at - created_at)) / 60) AS avg_cycle_time_minutes,
    COUNT(*) AS closed_case_count
FROM cases
WHERE organization_id = $1
  AND closed_at IS NOT NULL
  AND closed_at >= bucket_start
  AND closed_at < bucket_start + interval
```

**GetAgingCases:**
```sql
SELECT
    id AS case_id,
    case_number,
    status,
    created_at,
    EXTRACT(EPOCH FROM (NOW() - created_at)) / 3600 AS age_hours,
    service_type
FROM cases
WHERE organization_id = $1
  AND status NOT IN ('CLOSED', 'REJECTED')
  AND EXTRACT(EPOCH FROM (NOW() - created_at)) / 3600 > $2
ORDER BY age_hours DESC
```

**GetPendingReviews:**
```sql
SELECT
    COUNT(*) AS pending_reviews,
    AVG(EXTRACT(EPOCH FROM (NOW() - created_at)) / 60) AS avg_wait_minutes
FROM review_queue
WHERE organization_id = $1
  AND status = 'PENDING'
```

**GetDecisions:**
```sql
SELECT
    COUNT(*) AS total_decisions,
    COUNT(*) FILTER (WHERE decision = 'APPROVED') AS approved_count,
    COUNT(*) FILTER (WHERE decision = 'REJECTED') AS rejected_count,
    COUNT(*) FILTER (WHERE decision = 'PARTIAL') AS partial_count,
    CASE WHEN COUNT(*) > 0 THEN
        COUNT(*) FILTER (WHERE decision = 'APPROVED')::float / COUNT(*)
    ELSE 0 END AS approval_rate,
    AVG(EXTRACT(EPOCH FROM (completed_at - created_at)) / 60) AS avg_review_minutes
FROM decisions
WHERE organization_id = $1
  AND created_at >= bucket_start
  AND created_at < bucket_start + interval
```

**GetAssistanceOutcomes:**
```sql
SELECT
    COUNT(*) AS total_assistance,
    COUNT(*) FILTER (WHERE status = 'COMPLETED') AS completed_count,
    COUNT(*) FILTER (WHERE status = 'IN_PROGRESS') AS in_progress_count,
    CASE WHEN COUNT(*) > 0 THEN
        COUNT(*) FILTER (WHERE status = 'COMPLETED')::float / COUNT(*)
    ELSE 0 END AS completion_rate
FROM assistance
WHERE organization_id = $1
  AND created_at >= bucket_start
  AND created_at < bucket_start + interval
```

**GetEvidenceVerification:**
```sql
SELECT
    COUNT(*) AS total_evidence,
    COUNT(*) FILTER (WHERE verification_status = 'VERIFIED') AS verified_count,
    COUNT(*) FILTER (WHERE verification_status = 'REJECTED') AS rejected_count,
    COUNT(*) FILTER (WHERE verification_status = 'PENDING') AS pending_count,
    CASE WHEN COUNT(*) > 0 THEN
        COUNT(*) FILTER (WHERE verification_status = 'VERIFIED')::float / COUNT(*)
    ELSE 0 END AS verification_rate
FROM evidence
WHERE organization_id = $1
  AND created_at >= bucket_start
  AND created_at < bucket_start + interval
```

**GetInformationRequired:**
```sql
SELECT
    COUNT(*) AS total_requests,
    COUNT(*) FILTER (WHERE status = 'COMPLETED') AS provided_count,
    COUNT(*) FILTER (WHERE status = 'PENDING' AND due_at < NOW()) AS overdue_count,
    CASE WHEN COUNT(*) > 0 THEN
        COUNT(*) FILTER (WHERE status = 'COMPLETED')::float / COUNT(*)
    ELSE 0 END AS response_rate
FROM follow_ups
WHERE organization_id = $1
  AND type = 'INFORMATION_REQUIRED'
  AND created_at >= bucket_start
  AND created_at < bucket_start + interval
```

### 5.2 GetAllMetrics Aggregation

The `GetAllMetrics` method aggregates all individual metrics into a single `OperationsMetrics` response:

```go
func (s *operationsService) GetAllMetrics(ctx context.Context, orgID uuid.UUID, period MetricPeriod, bucket time.Time) (*OperationsMetrics, error) {
    metrics := &OperationsMetrics{
        OrganizationID: orgID,
        Period: period,
        Bucket: bucket,
    }

    var err error
    if metrics.CaseVolume, err = s.repo.GetCaseVolume(ctx, orgID, period, bucket); err != nil {
        return nil, err
    }
    if metrics.WorkflowThroughput, err = s.repo.GetWorkflowThroughput(ctx, orgID, period, bucket); err != nil {
        return nil, err
    }
    if metrics.StateDuration, err = s.repo.GetStateDuration(ctx, orgID, period, bucket); err != nil {
        return nil, err
    }
    if metrics.CaseCycleTime, err = s.repo.GetCaseCycleTime(ctx, orgID, period, bucket); err != nil {
        return nil, err
    }
    if metrics.AgingCases, err = s.repo.GetAgingCases(ctx, orgID, 720); err != nil {
        return nil, err
    }
    if metrics.PendingReviews, err = s.repo.GetPendingReviews(ctx, orgID); err != nil {
        return nil, err
    }
    if metrics.Decisions, err = s.repo.GetDecisions(ctx, orgID, period, bucket); err != nil {
        return nil, err
    }
    if metrics.AssistanceOutcomes, err = s.repo.GetAssistanceOutcomes(ctx, orgID, period, bucket); err != nil {
        return nil, err
    }
    if metrics.EvidenceVerification, err = s.repo.GetEvidenceVerification(ctx, orgID, period, bucket); err != nil {
        return nil, err
    }
    if metrics.InformationRequired, err = s.repo.GetInformationRequired(ctx, orgID, period, bucket); err != nil {
        return nil, err
    }

    return metrics, nil
}
```
